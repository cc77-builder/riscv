package xiangshan.backend.fu.wrapper

import chisel3._
import chisel3.util._
import org.chipsalliance.cde.config.Parameters
import xiangshan.ExceptionNO.illegalInstr
import xiangshan.backend.fu.{FuConfig, FuncUnit}
import xiangshan.backend.fu.vector.VdotExec
import xiangshan.backend.rob.RobPtr

object VdotOpType {
  def s8x8: UInt = 0.U
}

/** T06 adapter, compiled/tested through the real XiangShan interface.
  * T10 must supply exact decode, instruction-bound configuration, resolved v0
  * sources and a writeback/completion route. VS Off is rejected by DecodeUnit;
  * this interface has no VS input. Architectural Dirty is owned by retirement.
  */
class VdotFu(cfg: FuConfig)(implicit p: Parameters) extends FuncUnit(cfg) {
  require(cfg.fuType == xiangshan.backend.fu.FuType.vdot)
  require(cfg.destDataBits == 128 && cfg.vconfigIdx == 4 && cfg.maskSrcIdx == 3)
  require(cfg.latency.latencyVal.contains(1) && cfg.latency.extraLatencyVal.getOrElse(0) == 0)
  require(!cfg.writeVxsat && !cfg.writeVlRf && !cfg.writeFflags)

  private val exec = Module(new VdotExec((new RobPtr).getWidth))
  private val savedCtrl = Reg(chiselTypeOf(io.in.bits.ctrl))
  private val savedDebug = Reg(chiselTypeOf(io.in.bits.perfDebugInfo))
  private val vec = io.in.bits.ctrl.vpu.get

  // The core transports an operation, not the original instruction word. Exact
  // mask/match remains a decoder obligation; register fields are immaterial here.
  exec.io.in.bits.instruction := Mux(io.in.bits.ctrl.fuOpType === VdotOpType.s8x8,
    Mux(vec.vm, "h0200000b".U(32.W), "h0000000b".U(32.W)), 0.U(32.W))
  exec.io.in.bits.src1 := io.in.bits.data.src(0)
  exec.io.in.bits.src2 := io.in.bits.data.src(1)
  exec.io.in.bits.vl := io.in.bits.data.getSrcVConfig
  exec.io.in.bits.vtype := Cat(vec.vill, 0.U(55.W), vec.vma, vec.vta, vec.vsew, vec.vlmul)
  exec.io.in.bits.vstart := vec.vstart
  exec.io.in.bits.vs := 1.U // All non-Off states have identical execution behavior.
  exec.io.in.bits.tag := io.in.bits.ctrl.robIdx.asUInt

  // Use the existing circular ROB comparison, including wrap and flushItself.
  // pendingTag is deliberately unmasked by flush to avoid a combinational loop.
  private val pendingRob = exec.io.pendingTag.bits.asTypeOf(new RobPtr)
  exec.io.flush := exec.io.pendingTag.valid && pendingRob.needFlush(io.flush)
  private val incomingKilled = io.in.bits.ctrl.robIdx.needFlush(io.flush)
  exec.io.in.valid := io.in.valid && !incomingKilled
  io.in.ready := exec.io.in.ready && !incomingKilled
  exec.io.out.ready := io.out.ready
  io.out.valid := exec.io.out.valid

  when(io.in.fire) {
    savedCtrl := io.in.bits.ctrl
    savedDebug := io.in.bits.perfDebugInfo
    assert(!(io.in.bits.ctrl.vecWen.get && io.in.bits.ctrl.v0Wen.get),
      "vdot must select at most one vector destination")
  }

  io.out.bits := 0.U.asTypeOf(io.out.bits)
  io.out.bits.ctrl.robIdx := savedCtrl.robIdx
  io.out.bits.ctrl.pdest := savedCtrl.pdest
  io.out.bits.ctrl.vpu.get := savedCtrl.vpu.get
  io.out.bits.ctrl.vecWen.get := savedCtrl.vecWen.get && exec.io.out.bits.writeEnable
  io.out.bits.ctrl.v0Wen.get := savedCtrl.v0Wen.get && exec.io.out.bits.writeEnable
  io.out.bits.ctrl.exceptionVec.get(illegalInstr) := exec.io.out.bits.illegal
  io.out.bits.res.data := exec.io.out.bits.data
  io.out.bits.perfDebugInfo := savedDebug
  // Illegal requests still complete; out.valid must not depend on write enables.
  when(io.out.valid) {
    assert(exec.io.out.bits.tag === savedCtrl.robIdx.asUInt, "vdot metadata/result identity mismatch")
  }
}
